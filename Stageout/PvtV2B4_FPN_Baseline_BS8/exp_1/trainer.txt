# -*- coding: utf-8 -*-
"""
APBOXNet C32: 60 + 80 + 30

Stage I   : epoch 1-60
            Broad Exposure, Clean:Noisy = 1:1

Stage II  : epoch 61-140
            Reliability Annealing,
            Clean weight fixed at 1.0,
            Noisy weight cosine anneals 1.0 -> 0.0

Stage III : global epoch 141-170
            Promotion on CLEAN only.

Promotion modes:
    clean       : all Clean, no SBFT          (Stage-III control)
    acp         : hardest Clean subset, no SBFT
    sbft        : all Clean + SBFT             (isolates SBFT)
    acp_sbft    : hardest Clean subset + SBFT  (full promotion)

Optional PESF-style difficulty progression / Gaussian boundary perturbation
is kept in this trainer, but is OFF by default in the supplied C32 configs
so that the first 60+80+30 ablation remains clean.

The original repository's basemain.py is reused for:
    - ImageTrainDataset / ImageTestDataset
    - model registry
    - loss computation inside the model
    - optimizer/scaler helpers
    - COD evaluation metrics

No model file needs to be changed.
"""

# ============================================================
# 1) Reliability schedule: 60 + 80
# ============================================================

def get_reliability_weights(cfg, epoch):
    """
    Stage I:
        1-60    C:N weight = 1.0:1.0

    Stage II:
        61-140  C fixed at 1.0;
                N cosine 1.0 -> 0.0

    At epoch 140:
        C:N = 1.0:0.0
    """
    sch = cfg.train.curriculum.reliability

    clean_w = float(sch.get("clean_weight", 1.0))
    broad_end = int(sch.get("broad_end_epoch", 60))
    anneal_end = int(sch.get("anneal_end_epoch", 140))
    noisy_start = float(sch.get("noisy_start_weight", 1.0))
    noisy_end = float(sch.get("noisy_end_weight", 0.0))

    if epoch <= broad_end:
        return "stage1_broad", clean_w, noisy_start

    if epoch <= anneal_end:
        # x=0 at epoch 60, x=1 at epoch 140.
        x = (epoch - broad_end) / float(max(anneal_end - broad_end, 1))
        x = float(np.clip(x, 0.0, 1.0))

        # Smooth monotonic decrease:
        # epoch 60 -> noisy_start
        # epoch140 -> noisy_end
        coef = 0.5 * (1.0 + math.cos(math.pi * x))
        noisy_w = noisy_end + (noisy_start - noisy_end) * coef
        return "stage2_reliability_anneal", clean_w, float(noisy_w)

    return "stage2_clean_end", clean_w, noisy_end


def normalized_pool_probs(clean_weight, noisy_weight):
    total = float(clean_weight) + float(noisy_weight)
    if total <= 0:
        raise ValueError("clean_weight + noisy_weight must be > 0.")
    return float(clean_weight) / total, float(noisy_weight) / total

def _stem(path):
    return os.path.splitext(os.path.basename(path))[0]


def _read_difficulty_csv(path):
    """
    Expected CSV:
        name,difficulty

    Larger difficulty = harder.
    """
    scores = {}
    with open(path, "r", encoding="utf-8-sig", newline="") as f:
        reader = csv.DictReader(f)
        fields = reader.fieldnames or []

        name_key = next(
            (
                k for k in
                ("name", "image", "image_name", "filename", "file")
                if k in fields
            ),
            None,
        )

        if name_key is None or "difficulty" not in fields:
            raise KeyError(
                "difficulty CSV must contain a sample-name column "
                "and a 'difficulty' column."
            )

        for row in reader:
            key = core._normalize_sample_name(row[name_key])
            scores[key] = float(row["difficulty"])

    return scores


def build_difficulty_rank(dataset, cfg, pool_name):
    """
    Local dataset indices sorted Easy -> Hard.

    source='mask_area':
        smaller foreground -> harder
        runnable fallback only.

    source='csv':
        recommended for the paper; use your box-derived difficulty.
    """
    dcfg = cfg.train.curriculum.difficulty
    source = str(dcfg.get("source", "mask_area")).lower()

    scores = None
    if source == "csv":
        csv_path = core._resolve_existing_path(
            dcfg.get("csv", None),
            cfg=cfg,
        )
        scores = _read_difficulty_csv(csv_path)

    pairs = []
    missing = []

    for idx, (img_path, mask_path) in enumerate(dataset.total_data_paths):
        if source == "csv":
            name = core._normalize_sample_name(_stem(img_path))
            if name not in scores:
                missing.append(name)
                continue
            difficulty = float(scores[name])

        elif source == "mask_area":
            mask = cv2.imread(mask_path, cv2.IMREAD_GRAYSCALE)
            if mask is None:
                raise FileNotFoundError(mask_path)
            fg_ratio = float((mask > 0).mean())
            difficulty = 1.0 - fg_ratio

        else:
            raise ValueError(
                "difficulty.source must be 'mask_area' or 'csv'."
            )

        pairs.append((idx, difficulty))

    if missing:
        raise KeyError(
            f"[{pool_name}] difficulty CSV misses "
            f"{len(missing)} samples, examples={missing[:10]}"
        )

    pairs.sort(key=lambda x: x[1])  # Easy -> Hard

    if pairs:
        vals = np.asarray([v for _, v in pairs], dtype=np.float32)
        LOGGER.info(
            f"[Difficulty:{pool_name}] N={len(pairs)} "
            f"min={vals.min():.4f} "
            f"p50={np.median(vals):.4f} "
            f"p90={np.quantile(vals, 0.9):.4f} "
            f"max={vals.max():.4f}"
        )

    return [idx for idx, _ in pairs]


def get_difficulty_gate(cfg, epoch):
    """
    Optional PESF-style progressive Easy->Hard gate.

    IMPORTANT:
        Gaussian perturbs ONLY the difficulty boundary.
        It never changes Clean/Noisy reliability ratio.

    Return:
        base_fraction,
        perturbed_fraction,
        raw_gaussian_delta
    """
    dcfg = cfg.train.curriculum.difficulty

    if not bool(dcfg.get("enable", False)):
        return 1.0, 1.0, 0.0

    start_epoch = int(dcfg.get("start_epoch", 1))
    end_epoch = int(dcfg.get("end_epoch", 120))
    qmin = float(dcfg.get("min_fraction", 0.60))
    pacing = str(dcfg.get("pacing", "log")).lower()

    if epoch <= start_epoch:
        progress = 0.0
    elif epoch >= end_epoch:
        progress = 1.0
    else:
        e = epoch - start_epoch + 1
        E = end_epoch - start_epoch + 1

        if pacing == "log":
            progress = math.log(e + 1.0) / math.log(E + 1.0)
        elif pacing == "linear":
            progress = e / float(max(E, 1))
        elif pacing == "cosine":
            x = e / float(max(E, 1))
            progress = 0.5 - 0.5 * math.cos(math.pi * x)
        else:
            raise ValueError(
                f"Unknown difficulty pacing: {pacing}"
            )

    progress = float(np.clip(progress, 0.0, 1.0))
    base = qmin + (1.0 - qmin) * progress

    gaussian_on = bool(dcfg.get("gaussian_enable", False))
    if (not gaussian_on) or epoch >= end_epoch:
        return float(base), float(base), 0.0

    alpha0 = float(dcfg.get("gaussian_alpha", 0.10))
    gaussian_std = float(dcfg.get("gaussian_std", 0.10))

    anneal = max(
        0.0,
        1.0 - (epoch - start_epoch)
        / float(max(end_epoch - start_epoch, 1)),
    )
    alpha_e = alpha0 * anneal

    rng = np.random.RandomState(
        int(cfg.base_seed) + epoch * 7919
    )
    delta = float(
        rng.normal(loc=0.0, scale=gaussian_std)
    )

    fraction = float(
        np.clip(base + alpha_e * delta, qmin, 1.0)
    )
    return float(base), fraction, delta


def allowed_ranked_indices(ranked_local_indices, fraction, offset=0):
    n = len(ranked_local_indices)
    if n <= 0:
        return []
    k = max(
        1,
        int(math.ceil(n * float(fraction))),
    )
    return [
        int(i) + int(offset)
        for i in ranked_local_indices[:k]
    ]

# 3) Exact batch quota sampler
# ============================================================

class _PoolCycler:
    """
    Consume a shuffled pool without replacement until exhausted,
    then reshuffle. This is still replacement across a full epoch
    when requested sample mass exceeds pool size, but avoids repeated
    drawing of the same example before the pool has been traversed.
    """
    def __init__(self, indices, rng):
        self.indices = np.asarray(indices, dtype=np.int64)
        self.rng = rng
        self.buffer = np.empty((0,), dtype=np.int64)
        self.pos = 0
        self._refill()

    def _refill(self):
        if len(self.indices) == 0:
            self.buffer = np.empty((0,), dtype=np.int64)
        else:
            self.buffer = self.rng.permutation(self.indices)
        self.pos = 0

    def take(self, n):
        out = []

        while len(out) < n:
            if len(self.indices) == 0:
                raise RuntimeError("Trying to sample an empty pool.")

            if self.pos >= len(self.buffer):
                self._refill()

            available = len(self.buffer) - self.pos
            need = n - len(out)
            k = min(available, need)

            out.extend(
                self.buffer[self.pos:self.pos + k].tolist()
            )
            self.pos += k

        return out


class QuotaBatchSampler:
    """
    Exact batch-composition controller.

    BS=8 examples:
        noisy_prob = .50 -> 4C + 4N every batch
        noisy_prob = .25 -> 6C + 2N every batch

    For non-integer desired counts, floor/ceil compositions are distributed
    over the epoch so the average mass follows the continuous cosine schedule.
    """
    def __init__(
        self,
        clean_indices,
        noisy_indices,
        batch_size,
        num_samples,
        noisy_prob,
        seed,
    ):
        self.clean_indices = list(clean_indices)
        self.noisy_indices = list(noisy_indices)
        self.batch_size = int(batch_size)
        self.num_batches = int(num_samples) // self.batch_size
        self.noisy_prob = float(noisy_prob)
        self.seed = int(seed)

        if not self.clean_indices:
            raise RuntimeError("Clean allowed pool is empty.")

        if self.noisy_prob > 0 and not self.noisy_indices:
            raise RuntimeError(
                "Noisy probability > 0 but noisy allowed pool is empty."
            )

    def __len__(self):
        return self.num_batches

    def _noisy_count_sequence(self, rng):
        exact = self.batch_size * self.noisy_prob
        low = int(math.floor(exact))
        high = int(math.ceil(exact))

        if low == high:
            return [low] * self.num_batches

        frac = exact - low
        n_high = int(round(frac * self.num_batches))
        counts = (
            [high] * n_high
            + [low] * (self.num_batches - n_high)
        )
        rng.shuffle(counts)
        return counts

    def __iter__(self):
        rng = np.random.RandomState(self.seed)

        clean = _PoolCycler(self.clean_indices, rng)
        noisy = (
            _PoolCycler(self.noisy_indices, rng)
            if self.noisy_prob > 0
            else None
        )

        for n_noisy in self._noisy_count_sequence(rng):
            n_noisy = int(n_noisy)
            n_clean = self.batch_size - n_noisy

            batch = clean.take(n_clean)
            if n_noisy > 0:
                batch.extend(noisy.take(n_noisy))

            rng.shuffle(batch)
            yield batch

def prepare_curriculum_pools(cfg):
    pool_cfg = cfg.train.curriculum.pools

    for name in ("clean", "noisy"):
        if name not in pool_cfg:
            raise KeyError(
                f"train.curriculum.pools missing '{name}'."
            )

    clean_ds = core._build_train_subset(
        cfg=cfg,
        list_path=pool_cfg["clean"],
        pool_name="clean",
        augment=True,
    )
    noisy_ds = core._build_train_subset(
        cfg=cfg,
        list_path=pool_cfg["noisy"],
        pool_name="noisy",
        augment=True,
    )

    if len(clean_ds) == 0 or len(noisy_ds) == 0:
        raise RuntimeError(
            f"Empty pool: clean={len(clean_ds)}, noisy={len(noisy_ds)}"
        )

    diff_on = bool(
        cfg.train.curriculum.difficulty.get("enable", False)
    )

    if diff_on:
        clean_rank = build_difficulty_rank(
            clean_ds, cfg, "clean"
        )
        noisy_rank = build_difficulty_rank(
            noisy_ds, cfg, "noisy"
        )
    else:
        # All samples are allowed; rank order is irrelevant.
        clean_rank = list(range(len(clean_ds)))
        noisy_rank = list(range(len(noisy_ds)))

    LOGGER.info(
        f"Pools built: clean={len(clean_ds)}, noisy={len(noisy_ds)}"
    )

    return dict(
        clean=clean_ds,
        noisy=noisy_ds,
        concat=data.ConcatDataset([clean_ds, noisy_ds]),
        clean_rank=clean_rank,
        noisy_rank=noisy_rank,
        noisy_offset=len(clean_ds),
    )


def build_curriculum_loader(cfg, pools, epoch):
    stage, clean_w, noisy_w = get_reliability_weights(
        cfg, epoch
    )
    clean_p, noisy_p = normalized_pool_probs(
        clean_w, noisy_w
    )

    base_gate, gate, delta = get_difficulty_gate(
        cfg, epoch
    )

    clean_idx = allowed_ranked_indices(
        pools["clean_rank"],
        gate,
        offset=0,
    )
    noisy_idx = allowed_ranked_indices(
        pools["noisy_rank"],
        gate,
        offset=pools["noisy_offset"],
    )

    batch_sampler = QuotaBatchSampler(
        clean_indices=clean_idx,
        noisy_indices=noisy_idx,
        batch_size=cfg.train.batch_size,
        num_samples=cfg.train.curriculum.num_samples_per_epoch,
        noisy_prob=noisy_p,
        seed=int(cfg.base_seed) + epoch * 104729,
    )

    loader = data.DataLoader(
        dataset=pools["concat"],
        batch_sampler=batch_sampler,
        num_workers=cfg.train.num_workers,
        pin_memory=True,
        worker_init_fn=(
            core.pt_utils.customized_worker_init_fn
            if cfg.use_custom_worker_init
            else None
        ),
    )

    info = dict(
        stage=stage,
        clean_weight=clean_w,
        noisy_weight=noisy_w,
        clean_prob=clean_p,
        noisy_prob=noisy_p,
        difficulty_base=base_gate,
        difficulty_gate=gate,
        gaussian_delta=delta,
        clean_allowed=len(clean_idx),
        noisy_allowed=len(noisy_idx),
    )

    return loader, info


def append_sampling_log(path, epoch, info):
    os.makedirs(os.path.dirname(path), exist_ok=True)
    exists = os.path.isfile(path)

    fields = [
        "epoch",
        "stage",
        "clean_weight",
        "noisy_weight",
        "clean_prob",
        "noisy_prob",
        "difficulty_base",
        "difficulty_gate",
        "gaussian_delta",
        "clean_allowed",
        "noisy_allowed",
    ]

    with open(path, "a", encoding="utf-8", newline="") as f:
        writer = csv.DictWriter(
            f,
            fieldnames=fields,
        )
        if not exists:
            writer.writeheader()

        writer.writerow(
            dict(epoch=int(epoch), **info)
        )

# 5) Stage I + II training: global epoch 1-140

def train_curriculum_1_to_140(model, cfg):
    pools = prepare_curriculum_pools(cfg)

    loader_first, _ = build_curriculum_loader(
        cfg, pools, epoch=1
    )

    counter = core.recorder.TrainingCounter(
        epoch_length=len(loader_first),
        epoch_based=True,
        num_epochs=cfg.train.num_epochs,
        num_total_iters=None,
    )

    optimizer = core.pipeline.construct_optimizer(
        model=model,
        initial_lr=cfg.train.lr,
        mode=cfg.train.optimizer.mode,
        group_mode=cfg.train.optimizer.group_mode,
        cfg=cfg.train.optimizer.cfg,
    )

    scheduler = core.pipeline.Scheduler(
        optimizer=optimizer,
        num_iters=counter.num_total_iters,
        epoch_length=counter.num_inner_iters,
        scheduler_cfg=cfg.train.scheduler,
        step_by_batch=cfg.train.sche_usebatch,
    )

    scaler = core.pipeline.Scaler(
        optimizer,
        cfg.train.use_amp,
        set_to_none=cfg.train.optimizer.set_to_none,
    )

    loss_csv = os.path.join(
        cfg.path.pth_log,
        "loss_metrics_phase12.csv",
    )
    schedule_csv = os.path.join(
        cfg.path.pth_log,
        "sampling_schedule_phase12.csv",
    )

    LOGGER.info(
        "Stage I/II: 1-60 Broad 1:1; "
        "61-140 cosine noisy 1.0 -> 0.0."
    )
    LOGGER.info(
        "Batch quota ON. EMA/KD/saturation OFF."
    )

    for _ in range(counter.num_epochs):
        epoch = counter.curr_epoch + 1

        train_loader, sinfo = build_curriculum_loader(
            cfg, pools, epoch
        )
        append_sampling_log(
            schedule_csv, epoch, sinfo
        )

        model.train()

        if cfg.train.bn.freeze_status:
            core.pt_utils.frozen_bn_stats(
                model.encoder,
                freeze_affine=cfg.train.bn.freeze_affine,
            )

        sums = defaultdict(float)
        n_batches = 0

        for batch_idx, batch in enumerate(train_loader):
            scheduler.step(
                curr_idx=counter.curr_iter
            )

            batch_data = core.pt_utils.to_device(
                data=batch["data"],
                device=cfg.device,
            )

            with torch.cuda.amp.autocast(
                enabled=cfg.train.use_amp
            ):
                outputs = model(
                    data=batch_data,
                    iter_percentage=counter.curr_percent,
                )
                loss = outputs["loss"]

            items = core._loss_items_to_floats(
                outputs.get("loss_items", {})
            )
            bce = items.get(
                "bce",
                float(loss.detach().item()),
            )

            scaler.calculate_grad(
                loss=loss / cfg.train.grad_acc_step
            )
            if counter.every_n_iters(
                cfg.train.grad_acc_step
            ):
                scaler.update_grad()

            sums["total"] += float(
                loss.detach().item()
            )
            sums["bce"] += float(bce)
            n_batches += 1

            if cfg.log_interval > 0 and (
                batch_idx == 0
                or (batch_idx + 1) % cfg.log_interval == 0
                or batch_idx == len(train_loader) - 1
            ):
                LOGGER.info(
                    f"[{epoch:03d}] "
                    f"{batch_idx+1:04d}/{len(train_loader):04d} | "
                    f"C={sinfo['clean_prob']:.3f} "
                    f"N={sinfo['noisy_prob']:.3f} | "
                    f"D={sinfo['difficulty_gate']:.3f} | "
                    f"L={float(loss.detach().item()):.4f}"
                )

            if counter.is_last_total_iter():
                break

            counter.update_iter_counter()

        avg = core._append_epoch_losses(
            csv_path=loss_csv,
            epoch=epoch,
            stage_name=sinfo["stage"],
            loss_sums=sums,
            num_batches=n_batches,
        )

        core.io.save_weight(
            model=model,
            save_path=cfg.path.final_state_net,
        )

        should_val = (
            epoch >= int(cfg.train.val_start_epoch)
            and
            (
                epoch - int(cfg.train.val_start_epoch)
            ) % int(cfg.train.val_interval) == 0
        ) or epoch in (60, 140)

        if should_val:
            core.io.save_weight(
                model=model,
                save_path=os.path.join(
                    cfg.path.pth,
                    f"state_epoch_{epoch:03d}.pth",
                ),
            )

            old_save_results = cfg.save_results
            cfg.save_results = False

            core.test(
                model=model,
                cfg=cfg,
                epoch=epoch,
            )

            cfg.save_results = old_save_results

        LOGGER.info(
            f"Epoch {epoch:03d} done | "
            f"{sinfo['stage']} | "
            f"C={sinfo['clean_prob']:.3f} "
            f"N={sinfo['noisy_prob']:.3f} | "
            f"BCE={avg['bce']:.6f}"
        )

        counter.update_epoch_counter()

    return model

def _extract_eval_logits(output):
    """
    Original APBOXNet test() already expects model(data=...) to return
    a tensor in eval mode. Dict support is retained for robustness.
    """
    if torch.is_tensor(output):
        return output

    if isinstance(output, dict):
        for key in ("logits", "pred", "out"):
            value = output.get(key, None)
            if torch.is_tensor(value):
                return value

    raise TypeError(
        f"Cannot extract eval logits from {type(output)}"
    )


@torch.no_grad()
def compute_clean_difficulty(model, loader, device):
    """
    CurriSeg-style sample difficulty proxy:
        d_i = 1 - softIoU(prediction, reliable clean mask)

    Only CLEAN samples are ranked here.
    """
    was_training = model.training
    model.eval()

    pairs = []
    cursor = 0

    for batch in loader:
        batch_data = core.pt_utils.to_device(
            data=batch["data"],
            device=device,
        )

        logits = _extract_eval_logits(
            model(data=batch_data)
        )
        pred = logits.sigmoid()
        gt = (batch_data["mask"] > 0.5).to(
            pred.dtype
        )

        intersection = (
            pred * gt
        ).flatten(1).sum(dim=1)

        union = (
            pred + gt - pred * gt
        ).flatten(1).sum(dim=1)

        soft_iou = (
            intersection + 1e-6
        ) / (
            union + 1e-6
        )

        difficulty = 1.0 - soft_iou

        for value in difficulty.detach().cpu().tolist():
            pairs.append(
                (cursor, float(value))
            )
            cursor += 1

    if was_training:
        model.train()

    return pairs


def select_hard_clean_indices(score_pairs, hard_ratio):
    pairs = sorted(
        score_pairs,
        key=lambda x: x[1],
        reverse=True,
    )

    k = max(
        1,
        int(math.ceil(
            len(pairs) * float(hard_ratio)
        )),
    )

    values = np.asarray(
        [value for _, value in pairs],
        dtype=np.float32,
    )

    LOGGER.info(
        f"[ACP difficulty] clean N={len(pairs)}, "
        f"hard_ratio={float(hard_ratio):.3f}, "
        f"selected={k}, "
        f"p50={np.median(values):.4f}, "
        f"p90={np.quantile(values, 0.9):.4f}, "
        f"max={values.max():.4f}"
    )

    return [
        idx for idx, _ in pairs[:k]
    ]


# ============================================================
# 7) SBFT low-pass for ZoomNeXt/PVT multi-scale inputs
# ============================================================

def gaussian_kernel1d(
    kernel_size,
    sigma,
    device,
    dtype,
):
    x = torch.arange(
        kernel_size,
        device=device,
        dtype=dtype,
    ) - (kernel_size - 1) / 2.0

    kernel = torch.exp(
        -0.5 * (x / sigma) ** 2
    )

    return kernel / (
        kernel.sum() + 1e-8
    )


def lowpass_blur(
    images,
    kernel_size=11,
    sigma=3.0,
):
    if kernel_size % 2 == 0:
        kernel_size += 1

    _, channels, _, _ = images.shape

    k = gaussian_kernel1d(
        kernel_size=kernel_size,
        sigma=sigma,
        device=images.device,
        dtype=images.dtype,
    )

    kernel_y = k.view(
        1, 1, kernel_size, 1
    ).expand(
        channels, 1, kernel_size, 1
    )

    kernel_x = k.view(
        1, 1, 1, kernel_size
    ).expand(
        channels, 1, 1, kernel_size
    )

    pad = kernel_size // 2

    x = F.pad(
        images,
        (0, 0, pad, pad),
        mode="reflect",
    )
    x = F.conv2d(
        x,
        kernel_y,
        groups=channels,
    )

    x = F.pad(
        x,
        (pad, pad, 0, 0),
        mode="reflect",
    )
    x = F.conv2d(
        x,
        kernel_x,
        groups=channels,
    )

    return x


def _scaled_odd_kernel(base_kernel, scale):
    k = max(
        3,
        int(round(base_kernel * scale)),
    )
    if k % 2 == 0:
        k += 1
    return k


def apply_multiscale_sbft(
    batch_data,
    base_kernel=11,
    base_sigma=3.0,
):
    """
    Apply similar physical blur strength to S/M/L branches.

    image_s: ~0.5x
    image_m:  1.0x
    image_l: ~1.5x
    """
    for key, scale in (
        ("image_s", 0.5),
        ("image_m", 1.0),
        ("image_l", 1.5),
    ):
        if key not in batch_data:
            continue

        batch_data[key] = lowpass_blur(
            images=batch_data[key],
            kernel_size=_scaled_odd_kernel(
                base_kernel,
                scale,
            ),
            sigma=max(
                0.5,
                base_sigma * scale,
            ),
        )

    return batch_data


def get_sbft_probability(promotion_cfg, local_epoch):
    """
    Linear ramp is deliberately simple and interpretable.
    """
    total = int(
        promotion_cfg.get("epochs", 30)
    )

    p_start = float(
        promotion_cfg.get(
            "sbft_prob_start",
            0.10,
        )
    )
    p_end = float(
        promotion_cfg.get(
            "sbft_prob_end",
            0.50,
        )
    )

    if total <= 1:
        return p_end

    x = (
        local_epoch - 1
    ) / float(total - 1)

    x = float(
        np.clip(x, 0.0, 1.0)
    )

    return p_start + (
        p_end - p_start
    ) * x

def build_promotion_loader(model, cfg, mode):
    """
    mode:
        clean     -> all clean
        sbft      -> all clean
        acp       -> hard clean subset
        acp_sbft  -> hard clean subset
    """
    clean_list = (
        cfg.train.curriculum.pools["clean"]
    )

    # Stable unaugmented view for difficulty computation.
    eval_ds = core._build_train_subset(
        cfg=cfg,
        list_path=clean_list,
        pool_name="stage3_clean_eval",
        augment=False,
    )

    # Normal train augmentation for Stage III.
    train_ds = core._build_train_subset(
        cfg=cfg,
        list_path=clean_list,
        pool_name="stage3_clean_train",
        augment=True,
    )

    eval_names = [
        _stem(item[0])
        for item in eval_ds.total_data_paths
    ]
    train_names = [
        _stem(item[0])
        for item in train_ds.total_data_paths
    ]

    if eval_names != train_names:
        raise RuntimeError(
            "Stage-III clean eval/train order mismatch."
        )

    use_hard_subset = mode in (
        "acp",
        "acp_sbft",
    )

    if use_hard_subset:
        eval_loader = data.DataLoader(
            dataset=eval_ds,
            batch_size=int(
                cfg.train.promotion.get(
                    "eval_batch_size",
                    cfg.train.batch_size,
                )
            ),
            num_workers=cfg.train.num_workers,
            shuffle=False,
            drop_last=False,
            pin_memory=True,
        )

        scores = compute_clean_difficulty(
            model=model,
            loader=eval_loader,
            device=cfg.device,
        )

        hard_indices = select_hard_clean_indices(
            score_pairs=scores,
            hard_ratio=float(
                cfg.train.promotion.get(
                    "hard_ratio",
                    0.20,
                )
            ),
        )

        selected_ds = data.Subset(
            train_ds,
            hard_indices,
        )

        selected_count = len(hard_indices)

    else:
        selected_ds = train_ds
        selected_count = len(train_ds)

    loader = data.DataLoader(
        dataset=selected_ds,
        batch_size=cfg.train.batch_size,
        num_workers=cfg.train.num_workers,
        shuffle=True,
        drop_last=True,
        pin_memory=True,
        worker_init_fn=(
            core.pt_utils.customized_worker_init_fn
            if cfg.use_custom_worker_init
            else None
        ),
    )

    if len(loader) == 0:
        raise RuntimeError(
            "Stage-III loader is empty."
        )

    return loader, selected_count


def train_promotion_141_to_170(model, cfg):
    promotion = cfg.train.promotion

    if not bool(
        promotion.get("enable", True)
    ):
        LOGGER.info("Stage III disabled.")
        return model

    mode = str(
        promotion.get("mode", "acp")
    ).lower()

    valid_modes = {
        "clean",
        "acp",
        "sbft",
        "acp_sbft",
    }

    if mode not in valid_modes:
        raise ValueError(
            f"promotion.mode={mode} invalid; "
            f"use one of {sorted(valid_modes)}"
        )

    use_sbft = mode in (
        "sbft",
        "acp_sbft",
    )

    train_loader, selected_count = (
        build_promotion_loader(
            model=model,
            cfg=cfg,
            mode=mode,
        )
    )

    # Independent Stage-III optimizer.
    # This makes T0/T1/T2/T3 promotion ablations fair:
    # same optimizer reset, same LR, same number of local epochs.
    optimizer = core.pipeline.construct_optimizer(
        model=model,
        initial_lr=float(
            promotion.get("lr", 1e-5)
        ),
        mode=cfg.train.optimizer.mode,
        group_mode=cfg.train.optimizer.group_mode,
        cfg=cfg.train.optimizer.cfg,
    )

    scaler = core.pipeline.Scaler(
        optimizer,
        cfg.train.use_amp,
        set_to_none=cfg.train.optimizer.set_to_none,
    )

    n_epochs = int(
        promotion.get("epochs", 30)
    )
    global_start = (
        int(cfg.train.num_epochs) + 1
    )

    sbft_kernel = int(
        promotion.get("sbft_kernel", 11)
    )
    sbft_sigma = float(
        promotion.get("sbft_sigma", 3.0)
    )

    LOGGER.info(
        "=" * 70
    )
    LOGGER.info(
        f"Stage III START | "
        f"mode={mode} | "
        f"global={global_start}-"
        f"{global_start+n_epochs-1} | "
        f"selected_clean={selected_count} | "
        f"lr={float(promotion.get('lr', 1e-5)):.2e} | "
        f"SBFT={use_sbft}"
    )
    LOGGER.info(
        "=" * 70
    )

    loss_csv = os.path.join(
        cfg.path.pth_log,
        f"loss_metrics_stage3_{mode}.csv",
    )

    for local_epoch in range(
        1,
        n_epochs + 1,
    ):
        global_epoch = (
            global_start + local_epoch - 1
        )

        sbft_prob = (
            get_sbft_probability(
                promotion,
                local_epoch,
            )
            if use_sbft
            else 0.0
        )

        rng = np.random.RandomState(
            int(cfg.base_seed)
            + global_epoch * 65537
        )

        model.train()

        if cfg.train.bn.freeze_status:
            core.pt_utils.frozen_bn_stats(
                model.encoder,
                freeze_affine=cfg.train.bn.freeze_affine,
            )

        loss_sum = 0.0
        bce_sum = 0.0
        n_batches = 0

        for batch_idx, batch in enumerate(
            train_loader
        ):
            batch_data = core.pt_utils.to_device(
                data=batch["data"],
                device=cfg.device,
            )

            sbft_this_batch = (
                use_sbft
                and rng.rand() < sbft_prob
            )

            if sbft_this_batch:
                batch_data = apply_multiscale_sbft(
                    batch_data=batch_data,
                    base_kernel=sbft_kernel,
                    base_sigma=sbft_sigma,
                )

            with torch.cuda.amp.autocast(
                enabled=cfg.train.use_amp
            ):
                outputs = model(
                    data=batch_data,
                    iter_percentage=1.0,
                )
                loss = outputs["loss"]

            items = core._loss_items_to_floats(
                outputs.get(
                    "loss_items",
                    {},
                )
            )
            bce = items.get(
                "bce",
                float(loss.detach().item()),
            )

            scaler.calculate_grad(
                loss=loss
            )
            scaler.update_grad()

            loss_sum += float(
                loss.detach().item()
            )
            bce_sum += float(bce)
            n_batches += 1

            if cfg.log_interval > 0 and (
                batch_idx == 0
                or
                (batch_idx + 1) % cfg.log_interval == 0
                or
                batch_idx == len(train_loader) - 1
            ):
                LOGGER.info(
                    f"[StageIII {global_epoch:03d}] "
                    f"{batch_idx+1:04d}/{len(train_loader):04d} | "
                    f"mode={mode} | "
                    f"SBFT-p={sbft_prob:.3f} "
                    f"on={int(sbft_this_batch)} | "
                    f"L={float(loss.detach().item()):.4f}"
                )

        sums = {
            "total": loss_sum,
            "bce": bce_sum,
        }

        avg = core._append_epoch_losses(
            csv_path=loss_csv,
            epoch=global_epoch,
            stage_name=f"stage3_{mode}",
            loss_sums=sums,
            num_batches=n_batches,
        )

        LOGGER.info(
            f"[StageIII {global_epoch:03d}] "
            f"mode={mode} | "
            f"L={avg['total']:.6f} "
            f"BCE={avg['bce']:.6f} | "
            f"SBFT-p={sbft_prob:.3f}"
        )

        val_interval = int(
            promotion.get(
                "val_interval",
                5,
            )
        )

        if (
            local_epoch % val_interval == 0
            or local_epoch == n_epochs
        ):
            core.io.save_weight(
                model=model,
                save_path=os.path.join(
                    cfg.path.pth,
                    f"state_epoch_"
                    f"{global_epoch:03d}_"
                    f"{mode}.pth",
                ),
            )

            old_save_results = cfg.save_results
            cfg.save_results = False

            core.test(
                model=model,
                cfg=cfg,
                epoch=global_epoch,
            )

            cfg.save_results = old_save_results

    core.io.save_weight(
        model=model,
        save_path=cfg.path.final_state_net,
    )

    return model

def main():
    cfg = core.parse_cfg()

    # Keep the exact trainer snapshot used by this experiment.
    shutil.copy(
        __file__,
        cfg.path.trainer_copy,
    )

    core.pt_utils.initialize_seed_cudnn(
        seed=cfg.base_seed,
        deterministic=cfg.deterministic,
    )

    model_class = (
        core.model_zoo.__dict__.get(
            cfg.model_name
        )
    )

    assert model_class is not None, (
        "Please check --model-name"
    )

    model = model_class(
        num_frames=1,
        pretrained=cfg.pretrained,
        use_checkpoint=cfg.use_checkpoint,
    )

    LOGGER.info(
        inspect.getsource(model_class)
    )

    model.to(cfg.device)

    if cfg.load_from:
        core.io.load_weight(
            model=model,
            load_path=cfg.load_from,
            strict=True,
        )

    LOGGER.info(
        "Number of Parameters: "
        f"{sum(v.numel() for v in model.parameters())}"
    )

    if not cfg.evaluate:
        model = train_curriculum_1_to_140(
            model=model,
            cfg=cfg,
        )

        model = train_promotion_141_to_170(
            model=model,
            cfg=cfg,
        )

    if cfg.evaluate or cfg.has_test:
        core.io.save_weight(
            model=model,
            save_path=cfg.path.final_state_net,
        )

        core.test(
            model=model,
            cfg=cfg,
        )

    LOGGER.info("End training...")


if __name__ == "__main__":
    main()
